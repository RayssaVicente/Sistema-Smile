const prisma = new PrismaClient();

export async function handleFeedbackFlow(telefone: string, nomeCliente: string, mensagemRecebida: string, sendMessage: Function) {
    const textoLimpo = mensagemRecebida.trim();

    // Busca o paciente no banco pelo telefone para saber o status atual dele
    const patient = await prisma.patient.findUnique({
        where: { phone: telefone }
    });

    // Se o paciente não existe ou o status dele é IDLE (normal), o feedback não interfere em nada
    if (!patient || patient.status === 'IDLE' || !patient.status) {
        return false; // Retorna false para indicar que o fluxo de feedback ignorou a mensagem e o bot livre pode processar
    }

    // ETAPA 1: O BOT JÁ PEDIU A NOTA E ESTÁ AGUARDANDO A RESPOSTA NUMÉRICA
    if (patient.status === 'AGUARDANDO_NOTA') {
        const nota = parseInt(textoLimpo);

        if (isNaN(nota) || nota < 1 || nota > 5) {
            await sendMessage(telefone, "Por favor, responda apenas com um número de *1 a 5* estrelas para avaliarmos nosso atendimento.");
            return true; // Mensagem tratada pelo fluxo de feedback
        }

        await processarNota(patient.id, telefone, nomeCliente, nota, sendMessage);
        return true;
    }

    // ETAPA 2: O CLIENTE DEU NOTA BAIXA E ESTÁ DIGITANDO O COMENTÁRIO
    if (patient.status === 'AGUARDANDO_COMENTARIO') {
        // Podemos guardar a nota temporariamente no banco ou resgatar. Para simplificar, 
        // se preferir, podemos ler a última nota salva na tabela Feedback ou criar um campo 'tempNota' no Patient.
        // Vamos buscar a última nota registrada desse telefone que ainda não teve comentário ou salvar direto:
        
        const comentario = textoLimpo;

        // Atualiza o último feedback pendente ou cria um novo com o comentário
        const ultimoFeedback = await prisma.feedback.findFirst({
            where: { telefone },
            orderBy: { criadoEm : 'desc' }
        });

        if (ultimoFeedback) {
            await prisma.feedback.update({
                where: { id: ultimoFeedback.id },
                data: { comentario }
            });
        } else {
            await prisma.feedback.create({
                data: { telefone, nome: nomeCliente, nota: 1, comentario }
            });
        }

        // Reseta o status do paciente de volta para IDLE (liberando o chat normal)
        await prisma.patient.update({
            where: { id: patient.id },
            data: { status: 'IDLE' }
        });

        await sendMessage(
            telefone, 
            `Muito obrigado pelo seu feedback, ${nomeCliente}. Já repassamos sua mensagem para a nossa gestão para resolvermos isso o mais rápido possível. Pedimos desculpas por qualquer transtorno!`
        );

        console.log(`[ALERTA GESTÃO] Feedback de ${nomeCliente} (${telefone}). Comentário: "${comentario}"`);
        return true;
    }

    return false;
}

async function processarNota(patientId: string, telefone: string, nome: string, nota: number, sendMessage: Function) {
    if (nota >= 4) {
        // Nota Alta -> Salva e manda link do Google, liberando o paciente (status = IDLE)
        await prisma.feedback.create({
            data: { telefone, nome, nota, comentario: "Nota alta - Direcionado ao Google" }
        });

        await prisma.patient.update({
            where: { id: patientId },
            data: { status: 'IDLE' }
        });

        await sendMessage(
            telefone,
            `Ficamos extremamente felizes em saber disso! ⭐\n\nPoderia nos fazer um favorzinho e deixar essa mesma avaliação no nosso Google My Business para ajudar mais pessoas a sorrirem com a gente? Segue o link:\n👉 https://g.page/seu-link-aqui`
        );
    } else {
        // Nota Baixa -> Salva nota provisória e muda status para AGUARDANDO_COMENTARIO
        await prisma.feedback.create({
            data: { telefone, nome, nota, comentario: "Aguardando comentário de insatisfação" }
        });

        await prisma.patient.update({
            where: { id: patientId },
            data: { status: 'AGUARDANDO_COMENTARIO' }
        });

        await sendMessage(
            telefone,
            `Poxa, sentimos muito por isso. Queremos muito melhorar. O que aconteceu para você dar essa nota? Seu feedback é muito importante para nós. 💙`
        );
    }
}

import type { FastifyInstance } from 'fastify';
import { PrismaClient } from '@prisma/client';
import { getBotSettings, getHistoryForSecretary } from './configService.js';

const prisma = new PrismaClient();

export async function adminRoutes(app: FastifyInstance) {

    // Middleware de autenticação
    const adminAuth = async (request: any, reply: any) => {
        const authHeader = request.headers['authorization'];
        if (authHeader !== `Bearer ${process.env.SHARED_TOKEN}`) {
            return reply.status(401).send({ error: 'Unauthorized' });
        }
    };

    // 1. Buscar Configurações
    app.get('/admin/config', { preHandler: adminAuth }, async (request, reply) => {
        const settings = await getBotSettings();
        return reply.send(settings);
    });

    // 1.1 Atualizar Configurações (Ajustado para a tabela correta 'informacao')
    app.post('/admin/config', { preHandler: adminAuth }, async (request: any, reply) => {
        const { assistantNumber, evaluationLink } = request.body;
        
        // Verifique se sua tabela no prisma.schema se chama informacao ou botSettings
        const updated = await prisma.informacao.upsert({
            where: { id: 1 },
            update: { assistantNumber, evaluationLink },
            create: { id: 1, assistantNumber, evaluationLink, botNumber: process.env.BOT_NUMBER || "" }
        });

        return reply.send({ success: true, updated });
    });

    // 2. Histórico de Conversas
    app.get('/admin/historico/:phone', { preHandler: adminAuth }, async (request: any, reply) => {
        const { phone } = request.params;
        const historico = await getHistoryForSecretary(phone);
        return reply.send({ phone, historico });
    });

    // 3. Listar Agendamentos
    app.get('/admin/agendamentos', { preHandler: adminAuth }, async (request, reply) => {
        const agendamentos = await prisma.appointment.findMany({
            include: { patient: true },
            orderBy: { date: 'desc' },
            take: 50
        });
        return reply.send(agendamentos);
    });

    // 4. Listar Avaliações (Feedback)
    app.get('/admin/avaliacoes', { preHandler: adminAuth }, async (request, reply) => {
        const avaliacoes = await prisma.appointment.findMany({
            where: { status: 'CONCLUIDO' },
            include: { patient: true },
            orderBy: { date: 'desc' }
        });
        return reply.send(avaliacoes);
    });

    // 5. Disparar Cobrança Manual
    app.post('/admin/cobranca', { preHandler: adminAuth }, async (request: any, reply) => {
        const { telefone, nome, descricao, valor, vencimento, link_pagamento } = request.body;
        const limpaTelefone = telefone.replace(/\D/g, '');

        const mensagem = `Olá ${nome}, tudo bem?\n\n` +
                         `Passando para lembrar da sua cobrança: ${descricao}.\n` +
                         `Valor: R$ ${valor}\n` +
                         `Vencimento: ${vencimento}\n\n` +
                         `Clique aqui para pagar: ${link_pagamento}`;

        // Substitua pela sua função de envio se necessário
        return reply.send({ success: true, message: "Cobrança enviada com sucesso!" });
    });

    // 6. Finalizar Atendimento e Enviar Pesquisa
    app.post('/admin/finalizar', { preHandler: adminAuth }, async (request: any, reply) => {
        const { telefone } = request.body;
        const limpaTelefone = telefone.replace(/\D/g, '');

        const updated = await prisma.appointment.updateMany({
            where: { 
                patient: { phone: { contains: limpaTelefone } },
                status: { not: 'CONCLUIDO' }
            },
            data: { status: 'CONCLUIDO' }
        });

        if (updated.count > 0) {
            const settings = await getBotSettings();
            const linkAvaliacaoGoogle = settings.evaluationLink || "https://g.page/r/SEU_LINK/review";
            return reply.send({ success: true, message: `Atendimento de ${limpaTelefone} finalizado com sucesso.` });
        }

        return reply.status(404).send({ error: `Nenhuma consulta pendente encontrada para ${limpaTelefone}.` });
    });
}
